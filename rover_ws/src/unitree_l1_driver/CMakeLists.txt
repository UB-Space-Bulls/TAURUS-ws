cmake_minimum_required(VERSION 3.16)
project(unitree_l1_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This diagnostic targets Linux serial ports")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  set(SDK_ARCH x86_64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(SDK_ARCH aarch64)
else()
  message(FATAL_ERROR "SDK only includes x86_64 and aarch64 static libraries")
endif()
find_package(Threads REQUIRED)
add_executable(l1_sdk_reader src/sdk_reader.cpp src/sdk_serial_drain.cpp)
target_include_directories(l1_sdk_reader SYSTEM PRIVATE vendor/include)
target_link_libraries(l1_sdk_reader PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib/${SDK_ARCH}/libunitree_lidar_sdk.a Threads::Threads)
target_link_options(l1_sdk_reader PRIVATE "-Wl,--wrap=_ZN6serial6Serial12waitReadableEv")
add_executable(l1_wire_reader src/wire_reader.cpp)
target_include_directories(l1_wire_reader SYSTEM PRIVATE vendor/include)
option(L1_BUILD_ROS2 "Install as a ROS 2 ament package" OFF)
if(L1_BUILD_ROS2 OR DEFINED ENV{AMENT_PREFIX_PATH})
  find_package(ament_cmake REQUIRED)
  install(TARGETS l1_sdk_reader l1_wire_reader DESTINATION lib/${PROJECT_NAME})
  install(PROGRAMS scripts/l1_check.py scripts/l1_ros.py scripts/ros_topic_check.py DESTINATION lib/${PROJECT_NAME}
    PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(FILES scripts/l1_common.py DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch rviz docs DESTINATION share/${PROJECT_NAME})
  install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME health_logic COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -v)
  set_tests_properties(health_logic PROPERTIES ENVIRONMENT "L1_READER_DIR=$<TARGET_FILE_DIR:l1_sdk_reader>")
endif()
